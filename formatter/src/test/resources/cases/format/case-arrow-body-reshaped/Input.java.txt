class A {
    Object f(Mode mode) {
        return switch (mode) {
            case SOME_LONG_LABEL_NAME_ONE -> new TransferMeta(
                TransferType.FREE
            );
            case SOME_LONG_LABEL_NAME_TWO -> new TransferMeta(
                TransferType.FREE_TRANSFER,
                configuration.of("facilitiesFrequency")
                    .since(V2_2)
                    .asInt(3600),
                feedId
            );
            case ONE,
                TWO,
                THREE_LABEL_NAME_LONG -> new TransferMeta(
                TransferType.FREE_TRANSFER,
                Money.ZERO_USD_VALUE_LONG,
                payOnExit
            );
            default -> null;
        };
    }
}
